// Helper to generate realistic room electrical fixtures (Fans, Lights, CCTV, Projector, Speakers, AC, PCs, Electrical DB)
function makeFixtures(idPrefix, config) {
  const items = [];
  let idx = 1;
  const add = (name, kind, power_kw, start_hour, end_hour, count = 1, always24x7 = false) => {
    items.push({
      id: idPrefix * 100 + idx++,
      name,
      kind, // 'fan' | 'light' | 'projector' | 'speaker' | 'cctv' | 'ac' | 'computer' | 'electrical'
      power_kw,
      count,
      is_on: true,
      start_hour,
      end_hour,
      always24x7
    });
  };

  const s = config.start ?? 8;
  const e = config.end ?? 17;

  // Every room has Ceiling Fans, LED Tube Lights, CCTV Camera, and Smart Electrical DB Panel
  add(`Ceiling Fans x${config.fans || 4}`, 'fan', Number(((config.fans || 4) * 0.075).toFixed(2)), s, e, config.fans || 4);
  add(`LED Tube Lights x${config.lights || 6}`, 'light', Number(((config.lights || 6) * 0.04).toFixed(2)), s, e, config.lights || 6);
  add(`IP CCTV Dome Cam x${config.cctv || 2}`, 'cctv', Number(((config.cctv || 2) * 0.015).toFixed(2)), 0, 24, config.cctv || 2, true);
  add(`Smart DB & Meter Panel`, 'electrical', 0.05, 0, 24, 1, true);

  if (config.projector) {
    add(`HD Ceiling Projector`, 'projector', 0.35, s, e, 1);
  }
  if (config.speakers) {
    add(`PA Wall Speakers x${config.speakers}`, 'speaker', Number((config.speakers * 0.06).toFixed(2)), s, e, config.speakers);
  }
  if (config.acKw) {
    add(`Split AC Units (${config.acKw} kW)`, 'ac', config.acKw, s, e, Math.max(1, Math.round(config.acKw / 1.5)));
  }
  if (config.pcKw) {
    add(`${config.pcLabel || 'Lab Workstations'}`, 'computer', config.pcKw, s, e, config.pcCount || 25);
  }
  return items;
}

// ============================================================================
// PURE ENERGY LOGIC (Supports 24x7 critical gear like CCTV & Server Rooms)
// ============================================================================
function calcDailyKwh(eq) {
  if (!eq.is_on) return 0.0;
  return eq.power_kw * Math.max(0, eq.end_hour - eq.start_hour);
}

function calcWastedHours(eq, room) {
  if (!eq.is_on || eq.always24x7) return 0; // CCTV & Electrical DB panels are essential 24/7
  const onHours = Math.max(0, eq.end_hour - eq.start_hour);
  const overlap = Math.max(
    0,
    Math.min(eq.end_hour, room.occupied_to) - Math.max(eq.start_hour, room.occupied_from)
  );
  return Math.max(0, onHours - overlap);
}

function calcRoomStats(room) {
  const total = room.equipment.reduce((acc, e) => acc + calcDailyKwh(e), 0);
  const wasted = room.equipment.reduce((acc, e) => acc + e.power_kw * calcWastedHours(e, room), 0);
  const daily_kwh = Math.round(total * 100) / 100;
  const wasted_kwh = Math.round(wasted * 100) / 100;
  let status = "ok";
  if (wasted_kwh >= 18) status = "critical";
  else if (wasted_kwh >= 5) status = "warning";

  return {
    daily_kwh,
    wasted_kwh,
    status,
    daily_cost_inr: Math.round(daily_kwh * TARIFF_INR),
    wasted_cost_inr: Math.round(wasted_kwh * TARIFF_INR),
    monthly_wasted_inr: Math.round(wasted_kwh * TARIFF_INR * 30),
    monthly_wasted_co2: Math.round(wasted_kwh * CO2_PER_KWH * 30)
  };
}

function getRoomHourState(room, hour) {
  const isOccupied = hour >= room.occupied_from && hour < room.occupied_to;
  const activeEq = room.equipment.filter((e) => e.is_on && hour >= e.start_hour && hour < e.end_hour);
  const non24Active = activeEq.filter((e) => !e.always24x7);
  const currentKw = activeEq.reduce((s, e) => s + e.power_kw, 0);
  const isWastingNow = !isOccupied && non24Active.length > 0;
  return { isOccupied, activeEq, currentKw: Math.round(currentKw * 100) / 100, isWastingNow };
}

          acMeshes.push(acUnit);
        }

        // G. 3D COMPUTER WORKSTATIONS / LAB DESKS
        const hasPc = room.equipment.some((e) => e.kind === 'computer');
        if (hasPc) {
          const cols = Math.min(4, Math.max(2, Math.floor(rw / 1.4)));
          for (let c = 0; c < cols; c++) {
            const dx = (c - (cols - 1) / 2) * 1.1;
            const desk = new THREE.Mesh(
              new THREE.BoxGeometry(0.85, 0.45, 0.5),
              new THREE.MeshStandardMaterial({ color: 0x334155 })
            );
            desk.position.set(dx, 0.25, 0.2);
            fixturesGroup.add(desk);

            const monitorMat = new THREE.MeshStandardMaterial({
              color: 0x38bdf8,
              emissive: 0x0284c7,
              emissiveIntensity: 0.85
            });
            const monitor = new THREE.Mesh(
              new THREE.BoxGeometry(0.52, 0.34, 0.05),
              monitorMat
            );
            monitor.position.set(dx, 0.64, 0.15);
            fixturesGroup.add(monitor);
            pcScreenMeshes.push(monitor);
          }
        }

        // H. 3D ELECTRICAL DISTRIBUTION DB PANEL & SMART METER ON WALL
        const dbBox = new THREE.Mesh(
          new THREE.BoxGeometry(0.1, 0.42, 0.32),
          new THREE.MeshStandardMaterial({
            color: 0xf59e0b,
            emissive: 0xd97706,
            emissiveIntensity: 0.5
          })
        );
        dbBox.position.set(rw / 2 - 0.12, rh * 0.48, rd / 2 - 0.35);
        fixturesGroup.add(dbBox);

        roomObjectsRef.current[room.id] = {
          rGroup,
          floorTile,
          wallShell,
          edgeWire,
          fanSpinners,
          lightMeshes,
          projectorConeMeshes,
          cctvConeMeshes,
          speakerRingMeshes,
          acMeshes,
          pcScreenMeshes,
          worldPos: new THREE.Vector3(rx, floorData.level * FLOOR_HEIGHT + 1.2, rz)
        };
      });
    });

    const updateCamera = () => {
      const { radius, phi, theta } = sphericalRef.current;
      const t = targetRef.current;
      camera.position.x = t.x + radius * Math.sin(phi) * Math.sin(theta);
      camera.position.y = t.y + radius * Math.cos(phi);
      camera.position.z = t.z + radius * Math.sin(phi) * Math.cos(theta);
      camera.lookAt(t);
    };
    updateCamera();

    // Pointer & Raycaster Controls
    const raycaster = new THREE.Raycaster();
    const mouse = new THREE.Vector2();

    const getClickableMeshes = () => {
      const list = [];
      Object.values(roomObjectsRef.current).forEach((obj) => {
        if (obj.rGroup.parent && obj.rGroup.parent.visible) {
          list.push(obj.floorTile, obj.wallShell);
        }
      });
      return list;
    };

    const onPointerDown = (e) => {
      pointerRef.current = { isDown: true, startX: e.clientX, startY: e.clientY, moved: false };
    };

    const onPointerMove = (e) => {
      const rect = renderer.domElement.getBoundingClientRect();
      mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

      if (pointerRef.current.isDown) {
        const dx = e.clientX - pointerRef.current.startX;
        const dy = e.clientY - pointerRef.current.startY;
        if (Math.abs(dx) > 3 || Math.abs(dy) > 3) pointerRef.current.moved = true;

        sphericalRef.current.theta -= dx * 0.0075;
        sphericalRef.current.phi = Math.max(0.2, Math.min(Math.PI / 2.1, sphericalRef.current.phi - dy * 0.0065));
        pointerRef.current.startX = e.clientX;
        pointerRef.current.startY = e.clientY;
        updateCamera();
        return;
      }

      raycaster.setFromCamera(mouse, camera);
      const hits = raycaster.intersectObjects(getClickableMeshes(), false);
      if (hits.length > 0) {
        const data = hits[0].object.userData;
        renderer.domElement.style.cursor = 'pointer';
        setHoveredInfo({
          roomId: data.roomId,
          roomName: data.roomName,
          floorLevel: data.floorLevel,
          x: e.clientX - rect.left,
          y: e.clientY - rect.top
        });
      } else {
        renderer.domElement.style.cursor = 'grab';
        setHoveredInfo(null);
      }
    };

    const onPointerUp = (e) => {
      if (!pointerRef.current.isDown) return;
      const moved = pointerRef.current.moved;
      pointerRef.current.isDown = false;
      if (!moved) {
        const rect = renderer.domElement.getBoundingClientRect();
        mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
        mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
        raycaster.setFromCamera(mouse, camera);
        const hits = raycaster.intersectObjects(getClickableMeshes(), false);
        if (hits.length > 0) {
          onSelectRoom(hits[0].object.userData.roomId);
        }
      }
    };

    const onWheel = (e) => {
      e.preventDefault();
      sphericalRef.current.radius = Math.max(10, Math.min(72, sphericalRef.current.radius + e.deltaY * 0.03));
      updateCamera();
    };

    const dom = renderer.domElement;
    dom.addEventListener('pointerdown', onPointerDown);
    window.addEventListener('pointermove', onPointerMove);
    window.addEventListener('pointerup', onPointerUp);
    dom.addEventListener('wheel', onWheel, { passive: false });

    const onResize = () => {
      if (!container) return;
      camera.aspect = container.clientWidth / container.clientHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(container.clientWidth, container.clientHeight);
    };
    window.addEventListener('resize', onResize);

    // Animation Loop (Spins 3D Ceiling Fans & Pulses Speaker Sound Waves!)
    let frameId;
    const clock = new THREE.Clock();
    const animate = () => {
      frameId = requestAnimationFrame(animate);
      const elapsed = clock.getElapsedTime();

      spinningFansRef.current.forEach((item) => {
        if (item.rotor.userData.spinning) {
          item.rotor.rotation.y -= 0.18;
        }
      });

      speakerRingsRef.current.forEach((item) => {
        if (item.wave.userData.active) {
          const s = 1 + 0.45 * Math.sin(elapsed * 6);
          item.wave.scale.set(s, s, 1);
          item.wave.material.opacity = 0.25 + 0.35 * Math.sin(elapsed * 6);
        } else {
          item.wave.material.opacity = 0;
        }
      });

      renderer.render(scene, camera);
    };
    animate();

    return () => {
      cancelAnimationFrame(frameId);
      dom.removeEventListener('pointerdown', onPointerDown);
      window.removeEventListener('pointermove', onPointerMove);
      window.removeEventListener('pointerup', onPointerUp);
      dom.removeEventListener('wheel', onWheel);
      window.removeEventListener('resize', onResize);
      renderer.dispose();
    };
  }, []);

  // Synchronize Floor Visibility, Exploded Stacking, Room Status Colors & Live Appliance States!
  useEffect(() => {
    const BASE_H = 3.2;
    const EXPLODED_H = 6.2;

    floors.forEach((f) => {
      const fGroup = floorGroupsRef.current[f.level];
      if (!fGroup) return;

      if (activeFloorFilter === 'all') {
        fGroup.visible = true;
        fGroup.position.y = f.level * BASE_H;
      } else if (activeFloorFilter === 'exploded') {
        fGroup.visible = true;
        fGroup.position.y = f.level * EXPLODED_H;
      } else {
        fGroup.visible = f.level === Number(activeFloorFilter);
        fGroup.position.y = 0;
      }

      f.rooms.forEach((room) => {
        const obj = roomObjectsRef.current[room.id];
        if (!obj) return;

        const stats = calcRoomStats(room);
        const hourState = getRoomHourState(room, currentHour);
        const isSelected = room.id === selectedRoomId;

        // Color room floor tile & walls based on energy waste status
        let tileHex = 0x0d9488; // teal-emerald optimal
        let wallHex = 0x115e59;
        if (stats.status === 'critical') {
          tileHex = 0xdc2626;
          wallHex = 0x991b1b;
        } else if (stats.status === 'warning' || hourState.isWastingNow) {
          tileHex = 0xd97706;
          wallHex = 0x92400e;
        }

        obj.floorTile.material.color.setHex(tileHex);
        obj.floorTile.material.emissive.setHex(tileHex);
        obj.floorTile.material.emissiveIntensity = isSelected ? 0.85 : 0.35;

        obj.wallShell.material.color.setHex(wallHex);
        obj.wallShell.material.opacity = isSelected ? 0.42 : 0.22;

        obj.edgeWire.material.color.setHex(isSelected ? 0x22d3ee : 0xd6c79c);

        // Check each appliance state at currentHour
        const isEqActive = (kind) =>
          room.equipment.some(
            (e) => e.kind === kind && e.is_on && currentHour >= e.start_hour && currentHour < e.end_hour
          );

        const fansOn = isEqActive('fan');
        const lightsOn = isEqActive('light');
        const projOn = isEqActive('projector');
        const spkOn = isEqActive('speaker');
        const acOn = isEqActive('ac');
        const pcOn = isEqActive('computer');

        obj.fanSpinners.forEach((r) => {
          r.userData.spinning = fansOn;
        });

        obj.lightMeshes.forEach((m) => {
          m.material.color.setHex(lightsOn ? 0xfef9c3 : 0x334155);
          m.material.emissiveIntensity = lightsOn ? 1.0 : 0.0;
        });

        obj.projectorConeMeshes.forEach((m) => {
          m.visible = projOn;
        });

        obj.speakerRingMeshes.forEach((w) => {
          w.userData.active = spkOn;
        });

        obj.acMeshes.forEach((m) => {
          m.material.emissiveIntensity = acOn ? 0.65 : 0.0;
        });

        obj.pcScreenMeshes.forEach((m) => {
          m.material.color.setHex(pcOn ? 0x38bdf8 : 0x1e293b);
          m.material.emissiveIntensity = pcOn ? 0.9 : 0.0;
        });
      });
    });
  }, [floors, activeFloorFilter, selectedRoomId, currentHour]);

  // Camera Preset & Room Zoom Handler
  useEffect(() => {
    if (!cameraRef.current) return;
    if (cameraViewMode === 'courtyard') {
      // Looks across the central courtyard & amphitheater steps just like the uploaded photo!
      targetRef.current.set(5.5, 6.5, -2.5);
      sphericalRef.current = { radius: 22, phi: 1.18, theta: 0.02 };
    } else if (cameraViewMode === 'isometric') {
      targetRef.current.set(0, 4, 0);
      sphericalRef.current = { radius: 44, phi: 0.88, theta: 0.58 };
    } else if (cameraViewMode === 'top') {
      targetRef.current.set(0, 0, 0);
      sphericalRef.current = { radius: 42, phi: 0.18, theta: 0 };
    } else if (cameraViewMode === 'room_zoom' && selectedRoomId) {
      const obj = roomObjectsRef.current[selectedRoomId];
      if (obj) {
        const yPos = obj.rGroup.parent ? obj.rGroup.parent.position.y + 1.2 : 1.2;
        targetRef.current.set(obj.rGroup.position.x, yPos, obj.rGroup.position.z);
        sphericalRef.current = { radius: 12.5, phi: 0.72, theta: 0.45 };
      }
    }

    const { radius, phi, theta } = sphericalRef.current;
    const t = targetRef.current;
    cameraRef.current.position.set(
      t.x + radius * Math.sin(phi) * Math.sin(theta),
      t.y + radius * Math.cos(phi),
      t.z + radius * Math.sin(phi) * Math.cos(theta)
    );
    cameraRef.current.lookAt(t);
  }, [cameraViewMode, selectedRoomId, activeFloorFilter]);

  return (
    <div className="relative w-full h-full min-h-[520px] rounded-2xl overflow-hidden border border-slate-800 bg-[#050914]">
      <div ref={mountRef} className="w-full h-full min-h-[520px]" />

      {/* Department Signage Banner Overlay (Matches Courtyard Photo Banners) */}
      <div className="pointer-events-none absolute top-3 left-3 right-3 flex flex-wrap items-center justify-between gap-2">
        <div className="px-3.5 py-1.5 rounded-xl bg-slate-950/90 border border-cyan-500/40 backdrop-blur-md shadow-lg">
          <div className="text-[10px] font-bold uppercase tracking-widest text-cyan-400">
            {activeFloorFilter === 'all'
              ? 'ALL 5 FLOORS • CENTRAL COURTYARD & ATRIUM VIEW'
              : activeFloorFilter === 'exploded'
              ? 'EXPLODED 5-FLOOR ARCHITECTURAL STACK'
              : FLOOR_META[Number(activeFloorFilter)]?.banner}
          </div>
          <div className="text-xs text-slate-300 font-medium">
            3D Interior Fixtures Active: Spinning Ceiling Fans • LED Tubes • Projectors • CCTV • PA Speakers • Split ACs • Smart DB
          </div>
        </div>

        <div className="px-3 py-1.5 rounded-xl bg-slate-950/90 border border-slate-800 text-xs text-amber-300 font-semibold backdrop-blur-md">
          Drag to Orbit • Scroll to Zoom Into Rooms • Click Any Room
        </div>
      </div>

      {/* Hover Tooltip */}
      {hoveredInfo && (
        <div
          className="pointer-events-none absolute z-20 px-3 py-2 rounded-xl bg-slate-900/95 border border-cyan-400/50 text-xs shadow-xl backdrop-blur-md"
          style={{
            left: Math.min(hoveredInfo.x + 14, 520),
            top: Math.max(hoveredInfo.y - 44, 14)
          }}
        >
          <div className="font-bold text-white">{hoveredInfo.roomName}</div>
          <div className="text-cyan-300 text-[11px]">
            Floor {hoveredInfo.floorLevel} • Click to inspect Fans, Lights, CCTV & Projector
          </div>
        </div>
      )}

      {/* Bottom Legend */}
      <div className="pointer-events-none absolute bottom-3 left-3 flex flex-wrap items-center gap-3 px-3.5 py-2 rounded-xl bg-slate-950/90 border border-slate-800 text-[11px] backdrop-blur-md">
        <span className="flex items-center gap-1.5 text-emerald-400">
          <span className="w-2.5 h-2.5 rounded-full bg-emerald-400" /> Optimal Room
        </span>
        <span className="flex items-center gap-1.5 text-amber-400">
          <span className="w-2.5 h-2.5 rounded-full bg-amber-400" /> After-Hours Warning
        </span>
        <span className="flex items-center gap-1.5 text-red-400">
          <span className="w-2.5 h-2.5 rounded-full bg-red-500 animate-ping" /> Critical Wastage
        </span>
        <span className="text-slate-600">|</span>
        <span className="text-cyan-300">Cyan Cone = CCTV FOV / Projector Beam</span>
      </div>
    </div>
  );
}

        )}

        {/* ================================================================== */}
        {/* TAB 5: GEMINI AI ADVISOR */}
        {/* ================================================================== */}

                      className={`max-w-[85%] rounded-2xl px-4 py-3 text-xs whitespace-pre-wrap leading-relaxed ${

                onSubmit={(e) => {

                  type="submit"
                  className="px-4 py-2 rounded-xl bg-cyan-500 text-slate-950 font-bold text-xs"

              </form>
            </div>

            <div className="lg:col-span-4 p-5 rounded-2xl bg-slate-900/85 border border-slate-800 space-y-3">
              <div className="text-sm font-bold text-white">Quick Multi-Floor Prompts</div>
